inline std::optional<int> find_first_drop(const std::vector<SensorReading>& data) {
    for (const auto& r : data) {
        if (r.is_valid && r.voltage_pu < 0.90) {
            return r.time_ms; // Found it!
        }
    }
    return std::nullopt; // No data matched
}
