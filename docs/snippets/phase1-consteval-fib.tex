\begin{minted}[linenos,breaklines,frame=lines,framesep=2mm]{cpp}
consteval int compile_time_fibonacci(int n) {
    if (n <= 0) return 0;
    if (n == 1) return 1;
    int a = 0;
    int b = 1;
    for (int i = 2; i <= n; ++i) {
        int next = a + b;
        a = b;
        b = next;
    }
    return b;
}
\end{minted}
